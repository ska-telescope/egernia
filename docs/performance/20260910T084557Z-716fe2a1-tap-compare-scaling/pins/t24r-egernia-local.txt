bash -c set -e
if [ ! -s "$PGDATA/PG_VERSION" ]; then
  echo "cloning the primary into $PGDATA"
  rm -rf "$PGDATA"
  install -d -m 700 "$PGDATA"
  # Retried: a failed clone removes its own data directory, so a
  # second attempt starts clean, and three clones racing one primary
  # is a transient worth surviving rather than a run to restart.
  for attempt in 1 2 3; do
    pg_basebackup -h db -p 5432 -U tap -D "$PGDATA" -Fp -Xs -R -c fast -P && break
    echo "clone attempt $attempt failed; retrying in 10s"
    sleep 10
  done
  [ -s "$PGDATA/PG_VERSION" ] || { echo "could not clone the primary"; exit 1; }
fi
